Feed Analysis Simple

Analysis Overview

Research Question: Does feeding status (Fed vs. Unfed) affect the gut microbiome composition?

Statistical Note: We are analyzing the Sterile group at 15 days because it has a balanced sample size (4 Fed vs. 4 Unfed). The Muck group cannot be analyzed at this age due to insufficient samples.

Important: We have removed “Mother” from the statistical model because it is perfectly confounded with Feed status in this specific subset.

theme_set(theme_bw(base_size = 14))
# Load OTU Table
otu_raw <- read.table("kohler.opti_mcc.0.03.subsample.shared", header = TRUE, sep = "\t")
otu_mat <- otu_raw[, -(1:3)]
rownames(otu_mat) <- otu_raw$Group 
otu_mat <- t(otu_mat)
otu_tab <- otu_table(otu_mat, taxa_are_rows = TRUE)

# Load Taxonomy
tax_raw <- read.table("kohler.opti_mcc.0.03.cons.taxonomy", 
                      sep = "\t", header = TRUE, stringsAsFactors = FALSE, quote = "")
taxonomy_strings <- tax_raw$Taxonomy
names(taxonomy_strings) <- tax_raw$OTU
taxonomy_clean <- lapply(strsplit(taxonomy_strings, ";"), function(x) {
  x <- x[nchar(x) > 0]
  x <- gsub("\\(.*?\\)|\\\"", "", x)
  trimws(x)
})
max_ranks <- 7
taxonomy_mat <- do.call(rbind, lapply(taxonomy_clean, `length<-`, max_ranks))
colnames(taxonomy_mat) <- c("Kingdom", "Phylum", "Class", "Order", "Family", "Genus", "Species")
rownames(taxonomy_mat) <- names(taxonomy_strings)
tax_tab <- tax_table(as.matrix(taxonomy_mat))

# Load Metadata
Kohler.design <- read.table("Kohlerdesign.txt",
                            header = TRUE, sep = "\t", row.names = 1,
                            stringsAsFactors = TRUE, check.names = FALSE)

# Create Phyloseq
common_samples <- intersect(sample_names(otu_tab), rownames(Kohler.design))
otu_tab_pruned <- prune_samples(common_samples, otu_tab)
sample_data_obj <- sample_data(Kohler.design[common_samples, ])
physeq <- phyloseq(otu_tab_pruned, tax_tab, sample_data_obj)

# Rarefy
physeq_filt <- filter_taxa(physeq, function(x) sum(x > 3) > (0.2 * length(x)), TRUE)
min_depth <- min(sample_sums(physeq_filt))
set.seed(42)
physeq_rarefied <- rarefy_even_depth(physeq_filt, sample.size = min_depth, replace = FALSE)

# Alpha Diversity
alpha_diversity_rarefied <- estimate_richness(physeq_rarefied, measures = c("Observed", "Shannon", "Chao1"))
sample_metadata_rarefied <- as.data.frame(sample_data(physeq_rarefied))
alpha_data_rarefied <- cbind(alpha_diversity_rarefied, sample_metadata_rarefied)

Differential Abundance (DESeq2)

# Subset raw data
physeq_sterile_15_raw <- subset_samples(physeq, SampleType == "Sterile" & Age == "15+days")
physeq_sterile_15_raw <- prune_taxa(taxa_sums(physeq_sterile_15_raw) > 0, physeq_sterile_15_raw)

# DESeq2 (simplified model)
ds_obj_sterile <- phyloseq_to_deseq2(physeq_sterile_15_raw, ~ Feed)
ds_res_sterile <- DESeq(ds_obj_sterile, test = "Wald", fitType = "parametric")
res_sterile_feed <- results(ds_res_sterile, contrast = c("Feed", "Yes", "No"))

# Filter significant
sig_taxa_sterile <- as.data.frame(res_sterile_feed) %>%
  filter(padj < 0.2) %>%
  arrange(padj)

if (nrow(sig_taxa_sterile) > 0) {
  tax_info <- as.data.frame(tax_table(physeq)[rownames(sig_taxa_sterile), ])
  sig_taxa_sterile_annotated <- cbind(sig_taxa_sterile, tax_info)
  print(sig_taxa_sterile_annotated[, c("log2FoldChange", "padj", "Family", "Genus")])
} else {
  cat("\nNo taxa significantly different at FDR < 0.2\n")
}
